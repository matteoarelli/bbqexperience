    DECAY_MIN_IMP_28D = 200   # decay non significativo sotto questo

    def _fetch_clicks_per_week(days_back: int) -> dict[str, dict]:
        """Per ogni page URL, ritorna {clicks, impressions, ctr, position} sull'intervallo."""
        end = date.today() - timedelta(days=3)
        start = end - timedelta(days=days_back)
        rows = gsc_client.search_analytics(
            start_date=start.isoformat(), end_date=end.isoformat(),
            dimensions=["page"], row_limit=5000,
        )
        return {r["keys"][0]: r for r in rows}

    def _select_candidates() -> list[dict]:
        """Union decay + ctr_opportunity, dedup, top 10 by impressions.

        branches LISTED (not joined string) per supportare conditional prompt rendering
        in gsc_refresh_review._build_prompt (M8 fix).
        """
        last_7 = _fetch_clicks_per_week(7)
        last_28 = _fetch_clicks_per_week(28)
        candidates = {}
        for url, row28 in last_28.items():
            imp_28 = row28.get("impressions", 0)
            clicks_28 = row28.get("clicks", 0)
            ctr_28 = row28.get("ctr", 0)
            pos_28 = row28.get("position", 0)
            avg_week = clicks_28 / 4.0
            row7 = last_7.get(url, {})
            clicks_7 = row7.get("clicks", 0)
            branches: list[str] = []
            # Branch A: decay
            if imp_28 >= DECAY_MIN_IMP_28D and avg_week > 0 and clicks_7 < avg_week * DECAY_THRESHOLD:
                branches.append("decay")
            # Branch B: CTR opportunity — usa benchmark per posizione
            if imp_28 >= OPPORTUNITY_MIN_IMP_28D and 1 <= pos_28 <= 10:
                benchmark = benchmark_for_position(pos_28)

                    branches.append("ctr_opportunity")

                decay_pct = ((avg_week - clicks_7) / avg_week * 100) if avg_week > 0 else 0
                candidates[url] = {
                    "url": url,
                    "branches": branches,  # LIST per M8 conditional rendering
                    "metrics_28d": {"impressions": imp_28, "clicks": clicks_28, "ctr": ctr_28,
                                    "position": pos_28, "avg_clicks_per_week": avg_week},
                    "metrics_7d": {"clicks": clicks_7, "decay_pct": round(decay_pct, 1)},
                }
        ranked = sorted(candidates.values(), key=lambda c: c["metrics_28d"]["impressions"], reverse=True)
        return ranked[:TOP_N]

        Senza embeddings, semplice token overlap >=2 token significativi.

    def main():
        _sdxl_guard()
        print(f"[{datetime.now().isoformat()}] gsc_refresh avviato")
        STATE_DIR.mkdir(parents=True, exist_ok=True)
        # Reset queue file ogni run (settimanale, non append cumulativo)

        telegram.send_agent_report("GSC Refresh Review", msg, details=[next_step])

    if __name__ == "__main__": main()

    from agents.lib import gsc_client
    from datetime import date, timedelta
    # B1 fix: shared source of truth. NO try/except ImportError fallback.
    from agents.lib.ctr_benchmark import CTR_BENCHMARK, benchmark_for_position

    def get_gsc_digest() -> dict:
        """Digest GSC settimanale per il prompt strategist.

        4 sezioni: summary_delta (7d vs prev 7d), striking_top10 (pos 8-20),
        ctr_opportunity_top10 (pos 1-10 sotto benchmark), declining_top10 (decay >=30%).

        Graceful: ritorna dict con liste vuote su qualsiasi errore.
        """
        result = {
            "summary_delta": {}, "striking_top10": [],
            "ctr_opportunity_top10": [], "declining_top10": [],
        }
        try:
            end = date.today() - timedelta(days=3)
            start_7 = end - timedelta(days=7)
            start_prev = start_7 - timedelta(days=7)
            start_28 = end - timedelta(days=28)

            # Rows aggregate by date for 7d totals
            curr_7 = gsc_client.search_analytics(
                start_date=start_7.isoformat(), end_date=end.isoformat(),
                dimensions=["date"], row_limit=10)

                start_date=start_prev.isoformat(), end_date=(start_7 - timedelta(days=1)).isoformat(),
                dimensions=["date"], row_limit=10)

                if not rows: return {"clicks":0,"impressions":0,"ctr":0,"position":0}
