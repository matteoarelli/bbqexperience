    ```python
    #!/usr/bin/env python3
    """Meta Optimizer — Hetzner cron 03:30 UTC.

    Pesca articoli LIVE con impressions>=100/28gg e CTR sotto il benchmark
    per posizione * 0.6. Drafta nuovo seo_title/seo_description via Qwen
    con priming sulle top query GSC reali. Appende il proposal a
    state/meta_changes_pending.jsonl per il review Claude su Windows.

    NON applica direttamente: il review + Strapi PUT lo fa meta_review.py
    su Windows (Claude CLI non gira su Hetzner).

    Cron: 30 3 * * * (year-round, prima della finestra SDXL 04:00-05:00 UTC)
    Hour-check guard: skip se ora locale in [5,6,7].
    """
    import os, sys, io, json
    from datetime import datetime, timedelta, date
    from pathlib import Path

    # env loading (copy from content_generator.py lines 21-29 verbatim)
    # path insert agents/
    from agents.lib import gsc_client, strapi_client as strapi, telegram
    from agents.lib import claude_client as claude
    from agents.lib import atomic_io

    CTR_BENCHMARK = {1:0.25, 2:0.15, 3:0.09, 4:0.06, 5:0.045,
                     6:0.035, 7:0.025, 8:0.018, 9:0.015, 10:0.012}
    REWRITE_FACTOR = 0.6
    MIN_IMPRESSIONS_28D = 100
    RECENT_CHANGE_DAYS = 14  # skip se gia toccato negli ultimi N giorni

    STATE_DIR = Path(__file__).parent / "state"
    PENDING_FILE = STATE_DIR / "meta_changes_pending.jsonl"
    LOG_FILE = STATE_DIR / "meta_changes.jsonl"  # solo lettura qui — meta_review scrive
    PROMPT_TEMPLATE_PATH = Path(__file__).parent / "prompts" / "qwen_meta_draft.md"
    POWER_WORDS_IT_PATH = STATE_DIR / "power_words_it.txt"
    POWER_WORDS_ES_PATH = STATE_DIR / "power_words_es.txt"

    LOCALES = ["en", "it", "es"]
    LOCALE_FULL = {"en": "English", "it": "Italian", "es": "Spanish"}
    POWER_WORDS_EN = ["Ultimate","Complete","Best","Top 10","How to","Tested","Reviewed","Step-by-Step","Pitmaster's"]

    def _sdxl_guard():
        h = datetime.now().hour
        if 5 <= h <= 7:
            print(f"[meta_optimizer] skip: ora locale {h} dentro finestra SDXL")
            sys.exit(0)

    def _locale_from_url(url: str) -> str:
        for loc in LOCALES:
            if f"/{loc}/" in url: return loc
        return "en"

    def _content_type_from_url(url: str) -> str:
        # blog|tutorials|reviews|recipes — usa LOCALIZED_ROUTES inversa
        # ritorna lo strapi content type (blog-posts, tutorials, reviews, recipes)
        ...

    def _recent_changes_urls() -> set[str]:
        """Legge meta_changes.jsonl + meta_changes_pending.jsonl, ritorna URL toccati negli ultimi 14gg."""
        cutoff = datetime.now() - timedelta(days=RECENT_CHANGE_DAYS)
        urls = set()
        for p in [LOG_FILE, PENDING_FILE]:
            if not p.exists(): continue
            for line in p.read_text(encoding="utf-8").splitlines():
                try:
                    r = json.loads(line)
                    ts = datetime.fromisoformat(r.get("timestamp",""))
                    if ts >= cutoff: urls.add(r.get("url"))
                except: pass
        return urls

    def needs_meta_rewrite(row: dict) -> bool:
        impressions = row.get("impressions", 0)
        if impressions < MIN_IMPRESSIONS_28D: return False
        position = round(row.get("position", 99))
        if position < 1 or position > 10: return False
        benchmark = CTR_BENCHMARK[position]
        actual = row.get("ctr", 0)
        return actual < benchmark * REWRITE_FACTOR

    def _load_power_words(locale: str) -> str:
        if locale == "en": return ", ".join(POWER_WORDS_EN)
        p = POWER_WORDS_IT_PATH if locale == "it" else POWER_WORDS_ES_PATH
        if p.exists():
            return ", ".join([l.strip() for l in p.read_text(encoding="utf-8").splitlines() if l.strip()])
        return ""

    def _fetch_current_meta(content_type_strapi: str, slug: str, locale: str) -> dict | None:
        """Strapi find by slug+locale, return {documentId, seo_title, seo_description, excerpt}."""
        resp = strapi.find(content_type_strapi, locale=locale,
                           filters={"slug": {"$eq": slug}}, page_size=1,
                           fields=["seo_title","seo_description","excerpt","title"])
        items = resp.get("data", [])
        if not items: return None
        item = items[0]
        return {
            "documentId": item.get("documentId"),
            "current_title": item.get("seo_title") or item.get("title",""),
            "current_meta": item.get("seo_description","") or "",
            "excerpt": item.get("excerpt","") or "",
        }

    def build_prompt(row: dict, current: dict, queries: list[dict], locale: str) -> str:
        template = PROMPT_TEMPLATE_PATH.read_text(encoding="utf-8")
        url = row["keys"][0]
        position = row.get("position", 0)
        benchmark = CTR_BENCHMARK.get(round(position), 0.02)
        q_lines = "\n".join(f"- {q['keys'][0]} ({q.get('clicks',0)} clicks, {q.get('impressions',0)} imp, {q.get('ctr',0):.2%} CTR, pos {q.get('position',0):.1f})"
                            for q in queries[:5])
        return template.format(
            url=url, current_title=current["current_title"],
            current_meta_description=current["current_meta"], excerpt=current["excerpt"][:500],
            locale=locale, locale_full=LOCALE_FULL[locale],
            avg_position=position, ctr=row.get("ctr",0), benchmark_ctr=benchmark,
            impressions=row.get("impressions",0),
            top_queries_formatted=q_lines or "(nessuna query con clicks)",
            power_words=_load_power_words(locale),
        )

    def parse_qwen_output(raw: str) -> dict | None:
        """Parse 3-line output. Truncate title to 60 (last-space), meta to 155 (last-space)."""
        lines = [l.strip() for l in raw.strip().splitlines() if l.strip()]
        title = meta = reasoning = ""
        for l in lines:

        """Pesca query GSC in striking-distance: pos 8-20, imp>=30, ctr<benchmark.

        Pronta per merge con get_google_suggestions output dopo normalizzazione.

        results = []
        lo, hi = STRIKING_POSITION_RANGE
        for row in rows:

                "source": "gsc",
            })

        return results

        """Dedup via slugify. Source = "gsc+suggest" se entrambe contengono lo stesso slug."""
        out = {}

            key = slugify(s["query"])
            out[key] = {**s, "source": "suggest"}

    # env loading + import boilerplate
    from agents.lib import strapi_client as strapi, telegram, atomic_io
    from agents.claude_quality_gate import review_meta

    PENDING_FILE = Path(__file__).parent / "state" / "meta_changes_pending.jsonl"
    LOG_FILE = Path(__file__).parent / "state" / "meta_changes.jsonl"

    def _read_pending() -> list[dict]:
        if not PENDING_FILE.exists(): return []
        return [json.loads(l) for l in PENDING_FILE.read_text(encoding="utf-8").splitlines() if l.strip()]

    def _clear_pending(): PENDING_FILE.write_text("", encoding="utf-8")  # atomic via write_text

    def process_proposal(p: dict) -> dict:
        """Returns updated record with decision + applied flag."""
        result = review_meta(
            current_title=p["before"]["seo_title"], current_meta=p["before"]["seo_description"],
            proposed_title=p["proposed"]["seo_title"], proposed_meta=p["proposed"]["seo_description"],
            queries=p.get("query_targets", []), locale=p["locale"],
            excerpt="",  # excerpt non e nel pending — riprendilo da Strapi se serve
        )
        out = {**p, "review": {"decision": result.decision, "reasoning": result.reasoning,
                               "timestamp": datetime.now().isoformat()}}
        if result.decision == "approve":
            try:
                strapi.update(p["content_type"], p["documentId"],
                              {"seo_title": p["proposed"]["seo_title"],
                               "seo_description": p["proposed"]["seo_description"],
                               "slug": p["slug"]},  # slug obbligatorio per locale PUT Strapi v5
                              locale=p["locale"], skip_rebuild=True)
                out["applied"] = True
            except Exception as e:
                out["applied"] = False
                out["apply_error"] = str(e)
                out["review"]["decision"] = "apply_failed"
        else:
            out["applied"] = False
        return out

    def main():
        pending = _read_pending()
        if not pending:
            telegram.send_agent_report("Meta Review", "Nessuna proposta da revisionare oggi.")
            return
        applied = 0; rejected = 0; failed = 0
        for p in pending:
            result = process_proposal(p)
            atomic_io.append_jsonl_atomic(LOG_FILE, result)
            d = result["review"]["decision"]
            if d == "approve" and result.get("applied"): applied += 1
            elif d == "apply_failed": failed += 1
            else: rejected += 1
        _clear_pending()
        msg = f"Pending: {len(pending)} | Applicati: {applied} | Rifiutati: {rejected} | Fail: {failed}"
        next_step = (f"Verifica meta_changes.jsonl per i rejected (needs_human)." if rejected > 0
                     else "Tutti applicati con successo. Webhook rebuild suppresso.")
        telegram.send_agent_report("Meta Review", msg, details=[next_step])

    if __name__ == "__main__": main()
    ```
