    ```python
    def parse_qwen_output(raw: str) -> dict | None:
        """Parse 3-line output. Truncate title to 60 (last-space), meta to 155 (last-space)."""
        lines = [l.strip() for l in raw.strip().splitlines() if l.strip()]
        title = meta = reasoning = ""
        for l in lines:
            if l.startswith("SEO_TITLE:"): title = l.split(":",1)[1].strip()
            elif l.startswith("SEO_DESCRIPTION:"): meta = l.split(":",1)[1].strip()
            elif l.startswith("REASONING:"): reasoning = l.split(":",1)[1].strip()
        if not title or not meta: return None
        title = _truncate_word_boundary(title, 60)
        meta = _truncate_word_boundary(meta, 155)
        return {"seo_title": title, "seo_description": meta, "reasoning": reasoning}

    def _truncate_word_boundary(s: str, n: int) -> str:
        if len(s) <= n: return s
        cut = s[:n].rsplit(" ", 1)[0]
        return cut if cut else s[:n]

        print(f"  proposta accodata: {url} (pos {row.get('position',0):.1f} ctr {row.get('ctr',0):.2%})")
    ```

    ```python
    def scout_gsc_striking(days: int = 28) -> list[dict]:
        """Pesca query GSC in striking-distance: pos 8-20, imp>=30, ctr<benchmark.

        Ritorna lista di dict: {"query","clicks","impressions","ctr","position","source":"gsc"}
        Pronta per merge con get_google_suggestions output dopo normalizzazione.
        """
        try:
            from datetime import date, timedelta
            end = date.today() - timedelta(days=3)
            start = end - timedelta(days=days)
            rows = gsc_client.search_analytics(
                start_date=start.isoformat(), end_date=end.isoformat(),
                dimensions=["query"], row_limit=5000,
            )
        except Exception as e:
            print(f"[scout_gsc] GSC fetch fallita, skip striking source: {e}")
            return []
        results = []
        lo, hi = STRIKING_POSITION_RANGE
        for row in rows:
            pos = row.get("position", 0)
            imp = row.get("impressions", 0)
            ctr = row.get("ctr", 0)
            if not (lo <= pos <= hi): continue
            if imp < STRIKING_MIN_IMPRESSIONS: continue
            # skip se gia converte sopra benchmark (non e striking-distance vera)
            # benchmark_for_position clampa pos a 1..10, perfect per pos 8-20
            if ctr >= benchmark_for_position(pos): continue
            results.append({
                "query": row["keys"][0],
                "clicks": row.get("clicks",0),
                "impressions": imp, "ctr": ctr, "position": pos,
                "source": "gsc",
            })
        results.sort(key=lambda r: r["impressions"], reverse=True)
        return results

    def _dedup_candidates(suggest: list[dict], striking: list[dict]) -> list[dict]:
        """Dedup via slugify. Source = "gsc+suggest" se entrambe contengono lo stesso slug."""
        out = {}
        for s in suggest:
            key = slugify(s["query"])
            out[key] = {**s, "source": "suggest"}
        for g in striking:
            key = slugify(g["query"])
            if key in out:
                out[key]["source"] = "gsc+suggest"
                # mantieni metriche GSC se presenti
                for k in ("clicks","impressions","ctr","position"):
                    out[key][k] = g[k]
            else:
                out[key] = g
        return list(out.values())
    ```

    ```python
    #!/usr/bin/env python3
    """Meta Review — Windows Task Scheduler daily 09:00 local.

    Consuma state/meta_changes_pending.jsonl prodotto da meta_optimizer.py su Hetzner.
    Per ogni proposta: chiama Claude CLI quality gate; se approve -> strapi.update con
    skip_rebuild=True (webhook NON triggera Astro rebuild); se reject -> sposta in
    meta_changes.jsonl con decision='needs_human' per revisione manuale.

    Claude CLI gira solo su Windows (precedent: claude_review_runner.py).
    """
    # env loading + import boilerplate (mirror content_generator.py lines 21-29)
    from agents.lib import strapi_client as strapi, telegram, atomic_io
    from agents.claude_quality_gate import review_meta

    PENDING_FILE = Path(__file__).parent / "state" / "meta_changes_pending.jsonl"
    LOG_FILE = Path(__file__).parent / "state" / "meta_changes.jsonl"

    def _read_pending() -> list[dict]:
        if not PENDING_FILE.exists(): return []
        return [json.loads(l) for l in PENDING_FILE.read_text(encoding="utf-8").splitlines() if l.strip()]

    def _clear_pending(): PENDING_FILE.write_text("", encoding="utf-8")  # atomic via write_text

    def process_proposal(p: dict) -> dict:
        """Returns updated record with decision + applied flag."""
        result = review_meta(
            current_title=p["before"]["seo_title"], current_meta=p["before"]["seo_description"],
            proposed_title=p["proposed"]["seo_title"], proposed_meta=p["proposed"]["seo_description"],
            queries=p.get("query_targets", []), locale=p["locale"],
            excerpt="",  # excerpt non e nel pending — riprendilo da Strapi se serve
        )
        decision = "approve" if result.approved else "reject"
        out = {**p, "review": {"decision": decision, "reasoning": result.summary,
                               "timestamp": datetime.now().isoformat()}}
        if result.approved:
            try:
                strapi.update(p["content_type"], p["documentId"],
                              {"seo_title": p["proposed"]["seo_title"],
                               "seo_description": p["proposed"]["seo_description"],
                               "slug": p["slug"]},  # slug obbligatorio per locale PUT Strapi v5
                              locale=p["locale"], skip_rebuild=True)
                out["applied"] = True
            except Exception as e:
                out["applied"] = False
                out["apply_error"] = str(e)
                out["review"]["decision"] = "apply_failed"
        else:
            out["applied"] = False
        return out

    def main():
        pending = _read_pending()
        if not pending:
            telegram.send_agent_report("Meta Review", "Nessuna proposta da revisionare oggi.")
            return
        applied = 0; rejected = 0; failed = 0
        for p in pending:
            try:
                result = process_proposal(p)
                atomic_io.append_jsonl_atomic(LOG_FILE, result)
                d = result["review"]["decision"]
                if d == "approve" and result.get("applied"): applied += 1
                elif d == "apply_failed": failed += 1
                else: rejected += 1
            except Exception as e:
                failed += 1
                telegram.send_agent_report("Meta Review",
                    f"ERROR processing proposal: {e}",
                    details=["Verifica Claude CLI OAuth session", "Verifica state/meta_changes_pending.jsonl integrity"])
        _clear_pending()
        msg = f"Pending: {len(pending)} | Applicati: {applied} | Rifiutati: {rejected} | Fail: {failed}"
        next_step = (f"Verifica meta_changes.jsonl per i rejected (needs_human)." if rejected > 0
                     else "Tutti applicati con successo. Webhook rebuild suppresso.")
        telegram.send_agent_report("Meta Review", msg, details=[next_step])

    if __name__ == "__main__": main()
    ```
