    **STEP 2 (GREEN — structured-data.ts):**
    ```typescript
    /**
     * Structured Data Parsers — FAQ + HowTo + speakable per BBQ Experience.
     *
     * Schema.org markup ancora valido anche dopo deprecation rich results Google:
     * - FAQPage: deprecato Google 2026-05-07 ma ancora consumato da AI search
     *   (Perplexity, ChatGPT browse, Gemini, Bing).
     * - HowTo: deprecato 2023-09. Stesso razionale — AI engines consumano.
     * - speakable: BETA Google (originale Assistant). 2026: content-priority signal
     *   per AI Overviews. Usare selettori STRUTTURALI (mai class-based).
     *
     * Tutti i parser sono locale-aware (EN/IT/ES). Falliscono GRACEFULLY ritornando
     * null piuttosto che emettere schema sbagliato (better no schema than wrong).
     */

    export const FAQ_HEADING_PATTERNS = {
      en: /^##\s+(FAQ|Frequently Asked Questions)\s*$/im,
      it: /^##\s+Domande frequenti\s*$/im,
      es: /^##\s+Preguntas frecuentes\s*$/im,
    } as const;

    export const STEP_HEADING_PATTERNS = {
      en: /^#{2,3}\s+Step\s+(\d+)\s*[:.]?\s*(.*)$/gim,
      it: /^#{2,3}\s+Passo\s+(\d+)\s*[:.]?\s*(.*)$/gim,
      es: /^#{2,3}\s+Paso\s+(\d+)\s*[:.]?\s*(.*)$/gim,
    } as const;

    const OL_HOWTO_PATTERN = /<ol[^>]*>([\s\S]+?)<\/ol>/i;
    const LI_PATTERN = /<li[^>]*>([\s\S]+?)<\/li>/gi;

    export type Locale = 'en' | 'it' | 'es';
    export type Faq = { question: string; answer: string };
    export type FaqResult = { questions: Faq[] } | null;
    export type HowToStep = { name: string; text: string };
    export type HowToResult = { name: string; steps: HowToStep[] } | null;
    export type SpeakableSpec = { '@type': 'SpeakableSpecification'; cssSelector: string[] };

    /**
     * Estrae FAQ da markdown se trova heading "## FAQ" (o equivalente locale)
     * seguito da >= 2 coppie Q-A nel pattern "### Q\nA".
     * Ritorna null se < 2 coppie (single Q non e una FAQ).
     */
    export function detectFaqFromMarkdown(markdown: string, locale: Locale = 'en'): FaqResult {
      const pattern = FAQ_HEADING_PATTERNS[locale];
      const match = pattern.exec(markdown);
      if (!match) return null;
      // estrai sezione dopo heading FAQ fino al prossimo ##
      const startIdx = match.index + match[0].length;
      const restAfterFaq = markdown.slice(startIdx);
      const nextSection = /^##\s+/im.exec(restAfterFaq);
      const faqSection = nextSection ? restAfterFaq.slice(0, nextSection.index) : restAfterFaq;
      // parsa Q-A: ### question seguito da paragrafo fino al prossimo ### o fine
      const qaRegex = /^###\s+(.+?)\s*$([\s\S]+?)(?=^###\s+|\Z)/gim;
      const questions: Faq[] = [];
      let m: RegExpExecArray | null;
      while ((m = qaRegex.exec(faqSection)) !== null) {
        const q = m[1].trim();
        const a = m[2].trim();
        if (q && a) questions.push({ question: q, answer: a.replace(/\s+/g, ' ') });
      }
      return questions.length >= 2 ? { questions } : null;
    }

    /**
     * Estrae step HowTo da markdown. Due pattern:
     * 1. Heading "## Step N: title" (en) / "## Passo N: title" (it) / "## Paso N: title" (es)
     * 2. Ordered list <ol><li>...</li></ol> con >= 3 li
     * Ritorna null se < 3 step.
     */
    export function extractHowToSteps(
      markdown: string, locale: Locale = 'en', name: string = 'Tutorial',
    ): HowToResult {
      // Pattern 1: heading
      const headingPattern = STEP_HEADING_PATTERNS[locale];
      const steps: HowToStep[] = [];
      let m: RegExpExecArray | null;
      while ((m = headingPattern.exec(markdown)) !== null) {
        const stepNum = m[1];
        const stepTitle = (m[2] || '').trim();
        // body fino al prossimo step heading o ##
        const startIdx = m.index + m[0].length;
        const rest = markdown.slice(startIdx);
        const nextStep = /^#{2,3}\s+/m.exec(rest);
        const body = nextStep ? rest.slice(0, nextStep.index) : rest;
        steps.push({
          name: stepTitle || `Step ${stepNum}`,
          text: body.trim().replace(/\s+/g, ' ').slice(0, 500),
        });
      }
      if (steps.length >= 3) return { name, steps };
      // Pattern 2: <ol><li>
      const olMatch = OL_HOWTO_PATTERN.exec(markdown);
      if (olMatch) {
        const items: HowToStep[] = [];
        let im: RegExpExecArray | null;
        while ((im = LI_PATTERN.exec(olMatch[1])) !== null) {
          const text = im[1].replace(/<[^>]+>/g, '').trim();
          if (text) items.push({ name: `Step ${items.length + 1}`, text: text.slice(0, 500) });
        }
        if (items.length >= 3) return { name, steps: items };
      }
      return null;
    }

    /**
     * Costruisce SpeakableSpecification con selettori STRUTTURALI (mai class-based).
     * Sopravvive a refactor CSS Astro (auto-scoped class names cambiano).
     */
    export function buildSpeakableSpec(): SpeakableSpec {
      return {
        '@type': 'SpeakableSpecification',
        cssSelector: ['article > p:first-of-type', 'article h2'],
      };
    }
    ```

        # Heading pattern
        for h in soup.find_all(["h2", "h3"]):
            text = h.get_text(strip=True)
            if any(_re.match(p, text) for p in HOWTO_STEP_PATTERNS):
                return True
        # OL with 3+ items
        for ol in soup.find_all("ol"):
            if len(ol.find_all("li", recursive=False)) >= 3:
                return True
        return False

    def _jsonld_blocks(soup) -> list[dict]:
        out = []
        for s in soup.find_all("script", {"type": "application/ld+json"}):
            try:
                out.append(_json.loads(s.string or "{}"))
            except Exception:
                continue
        return out

        for b in blocks:
            if b.get("@type") == t: return True
            # @graph variant
            for item in b.get("@graph", []) or []:
                if isinstance(item, dict) and item.get("@type") == t: return True
