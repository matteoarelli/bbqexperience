       ```typescript
       let experimentDocId: string | null = null;

       if (!Astro.locals.isBot) {
         try {
           const experiments = await fetchCollection<StrapiAbExperiment>('ab-experiments', {
             locale,
             filters: {
               blog_post: { documentId: { $eq: post.documentId } },
               status: { $eq: 'active' },
             },
             populate: '*',
             pageSize: 1,
           });
           const experiment = experiments?.data?.[0];
           if (experiment) {
             experimentDocId = experiment.documentId;
             const variantCount = experiment.variant_c ? 4 : 3; // control + a + b [+ c]
             const variantIndex = assignVariant(Astro.locals.abId, post.documentId, variantCount);
             activeVariant = VARIANT_NAMES[variantIndex];
             // Mappa indice variante al testo corrispondente
             if (variantIndex === 1) displayTitle = experiment.variant_a;
             else if (variantIndex === 2) displayTitle = experiment.variant_b;
             else if (variantIndex === 3 && experiment.variant_c) displayTitle = experiment.variant_c;
             // variantIndex === 0 -> control -> titolo originale (displayTitle resta post.title)
           }
         } catch (e) {
           // Fallback silenzioso: mostra titolo originale se Strapi non risponde
           console.error('[A/B] Errore lookup esperimento:', e);
         }
       }
       ```
