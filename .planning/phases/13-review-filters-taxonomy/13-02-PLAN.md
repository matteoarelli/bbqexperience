import { parseFilterParams, hasActiveFilters, buildStrapiFilters, computeFacetCounts, buildFilterUrl, PRICE_BUCKETS, PRICE_BUCKET_LABELS, SCORE_THRESHOLDS } from '@lib/filters';

const t = (key: string) => getTranslation(translations, key);

// Parsing filtri dalla URL
const activeFilters = parseFilterParams(Astro.url.searchParams);
const filtersActive = hasActiveFilters(activeFilters);

// Paginazione e ordinamento
const currentPage = Math.max(1, parseInt(Astro.url.searchParams.get('page') || '1', 10));
const activeSort = Astro.url.searchParams.get('sort') || 'newest';
const sortMap = { newest: 'published_date:desc', highestScore: 'score_overall:desc', lowestScore: 'score_overall:asc' };

// Fetch TUTTE le review (campi minimi) per conteggi facet
const allReviewsResponse = await fetchCollection<StrapiReview>('reviews', {
  locale,
  populate: ['product', 'product.brand_relation', 'product.product_category'],
  fields: ['score_overall'],
  pageSize: 100,
});
const facetCounts = computeFacetCounts(allReviewsResponse.data);

// Fetch review filtrate per la pagina corrente
const strapiFilters = buildStrapiFilters(activeFilters);
const response = await fetchCollection<StrapiReview>('reviews', {
  locale,
  populate: ['product', 'product.brand_relation', 'product.product_category', 'gallery'],
  filters: strapiFilters,
  sort: sortMap[activeSort] || sortMap.newest,
  pageSize: 12,
  page: currentPage,
});
const reviews = response.data;
const totalCount = response.meta.pagination.total;
const totalPages = response.meta.pagination.pageCount;

// Fetch brand e categorie per le label dei filtri
const brandsResponse = await fetchCollection<StrapiBrand>('brands', { locale, pageSize: 100 });
const brandsWithProducts = brandsResponse.data
  .filter(b => facetCounts.brands[b.slug] > 0)
  .map(b => ({ slug: b.slug, name: b.name, count: facetCounts.brands[b.slug] }))
  .sort((a, b) => b.count - a.count);

const categoriesResponse = await fetchCollection<StrapiProductCategory>('product-categories', { locale, pageSize: 100 });
const categoriesWithProducts = categoriesResponse.data
  .filter(c => facetCounts.categories[c.slug] > 0)
  .map(c => ({ slug: c.slug, name: c.name, count: facetCounts.categories[c.slug] }))
  .sort((a, b) => b.count - a.count);

const reviewRoute = localizedRoutes.reviews[locale];
const basePath = `/${locale}/${reviewRoute}/`;

// Pre-compute URL per ogni opzione filtro (passati al drawer come props)
const brandsWithUrls = brandsWithProducts.map(b => ({
  ...b,
  url: buildFilterUrl(basePath, { ...activeFilters, brand: b.slug })
}));
const categoriesWithUrls = categoriesWithProducts.map(c => ({
  ...c,
  url: buildFilterUrl(basePath, { ...activeFilters, category: c.slug })
}));

// Dati per il drawer mobile
const priceBucketOptions = Object.entries(PRICE_BUCKETS)
  .filter(([key]) => (facetCounts.priceBuckets[key] || 0) > 0)
  .map(([key, _]) => ({
    key,
    label: PRICE_BUCKET_LABELS[key][locale],
    count: facetCounts.priceBuckets[key],
    url: buildFilterUrl(basePath, { ...activeFilters, price: key })
  }));

const scoreOptions = SCORE_THRESHOLDS.map(st => ({
  value: st.value,
  label: st.label,
  count: facetCounts.scoreThresholds[st.value] || 0,
  url: buildFilterUrl(basePath, { ...activeFilters, score: st.value })
}));

const clearUrl = buildFilterUrl(basePath, {});
