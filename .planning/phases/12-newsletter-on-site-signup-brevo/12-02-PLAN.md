```html
<script>
  // Gestione exit-intent modal — solo desktop con mouse reale
  (function initExitModal() {
    const modal = document.querySelector('[data-exit-modal]') as HTMLElement | null;
    if (!modal) return;

    const overlay = modal.querySelector('[data-exit-modal-overlay]') as HTMLElement;
    const card = modal.querySelector('[role="dialog"]') as HTMLElement;
    const closeBtn = modal.querySelector('[data-exit-modal-close]') as HTMLElement;
    const announce = modal.querySelector('[data-exit-modal-announce]') as HTMLElement;

    // D-09: Solo dispositivi con mouse reale
    if (!window.matchMedia('(hover: hover) and (pointer: fine)').matches) return;

    // Controlla soppressione
    const COOKIE_NAME = 'bbq-exit-modal-dismissed';
    const SESSION_KEY = 'bbq-exit-modal-shown';
    const SUBSCRIBED_KEY = 'bbq-newsletter-subscribed';

    function getCookie(name: string): string | null {
      const match = document.cookie.match(new RegExp(`(?:^|; )${name}=([^;]*)`));
      return match ? match[1] : null;
    }

    function setCookie(name: string, value: string, days: number): void {
      const expires = new Date(Date.now() + days * 864e5).toUTCString();
      document.cookie = `${name}=${value}; expires=${expires}; path=/; SameSite=Lax`;
    }

    if (getCookie(COOKIE_NAME) || sessionStorage.getItem(SESSION_KEY) || localStorage.getItem(SUBSCRIBED_KEY)) return;

    let previousFocus: HTMLElement | null = null;

    async function showModal(): void {
      if (modal.style.display !== 'none') return; // Gia aperto
      sessionStorage.setItem(SESSION_KEY, '1');
      previousFocus = document.activeElement as HTMLElement;

      modal.style.display = '';
      modal.setAttribute('aria-hidden', 'false');
      announce.textContent = modal.querySelector('[role="dialog"]')?.getAttribute('aria-label') || '';

      // Animazione GSAP
      const { gsap } = await import('gsap');
      gsap.fromTo(overlay, { opacity: 0 }, { opacity: 1, duration: 0.3, ease: 'power2.out' });
      gsap.fromTo(card, { scale: 0.9, opacity: 0 }, { scale: 1, opacity: 1, duration: 0.4, ease: 'back.out(1.4)', delay: 0.1, onComplete: () => {
        // Focus trap: primo elemento focusabile
        const focusable = card.querySelectorAll<HTMLElement>('input:not([tabindex="-1"]), button:not([disabled]), [tabindex]:not([tabindex="-1"])');
        focusable[0]?.focus();
      }});
    }

    async function closeModal(): void {
      setCookie(COOKIE_NAME, '1', 14); // 14 giorni soppressione
      const { gsap } = await import('gsap');
      gsap.to(card, { scale: 0.95, opacity: 0, duration: 0.2, ease: 'power2.in' });
      gsap.to(overlay, { opacity: 0, duration: 0.25, ease: 'power2.in', delay: 0.05, onComplete: () => {
        modal.style.display = 'none';
        modal.setAttribute('aria-hidden', 'true');
        announce.textContent = '';
        previousFocus?.focus();
      }});
    }

    // D-05: mouseleave solo quando clientY < 0 (uscita verso browser chrome)
    document.documentElement.addEventListener('mouseleave', (e: MouseEvent) => {
      if (e.clientY < 0) showModal();
    });

    // Chiudi su click overlay
    overlay.addEventListener('click', closeModal);

    // Chiudi su click X
    closeBtn.addEventListener('click', closeModal);

    // Keyboard: Escape chiude, Tab intrappola focus
    card.addEventListener('keydown', (e: KeyboardEvent) => {
      if (e.key === 'Escape') {
        e.preventDefault();
        closeModal();
        return;
      }
      if (e.key === 'Tab') {
        const focusable = card.querySelectorAll<HTMLElement>('input:not([tabindex="-1"]), button:not([disabled]), [tabindex]:not([tabindex="-1"])');
        const first = focusable[0];
        const last = focusable[focusable.length - 1];
        if (e.shiftKey && document.activeElement === first) {
          e.preventDefault();
          last.focus();
        } else if (!e.shiftKey && document.activeElement === last) {
          e.preventDefault();
          first.focus();
        }
      }
    });
  })();
</script>
```
